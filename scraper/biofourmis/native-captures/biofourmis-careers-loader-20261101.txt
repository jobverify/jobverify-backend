// Triggered in finally above
async function writeJobs(jobs) {
  loadingText.style.display = "block";
  jobsWrapper.innerHTML = "";
  jobs.forEach((job) => {
    renderJobs(job.name, job.jobs);
  });
  loadingText.style.display = "none";
}

function renderJobs(departmentName, jobs) {
  const jobWrapperNode = document.createElement("div");
  jobWrapperNode.classList.add("job-department");
  // jobWra
  jobWrapperNode.innerHTML = `<div class="margin-bottom margin-small">
            <h3>${departmentName}</h3>
          </div>`;
  jobs.forEach((job) => {
    const jobNode = document.createElement("a");
    jobNode.href = `${job.absolute_url}`;
    jobNode.setAttribute("job-location", job.location.name.trim());
    jobNode.className = "job-box w-inline-block";
    // jobNode.classList.add("job-box w-inline-block");
    jobNode.innerHTML = `<div class="margin-bottom margin-small">
              <div style="display:none;" class="text-size-small text-weight-semibold is-gray">
                ${departmentName}
              </div>
            </div>
            <div class="margin-bottom margin-small">
              <h4>${job.title}</h4>
            </div>
            <div class="job-location-text text-color-grey">${job.location.name}</div>`;

    jobWrapperNode.append(jobNode);
  });
  jobsWrapper.appendChild(jobWrapperNode);
}

function renderDepartment(department) {
  let departmentNode = document.createElement("option");
  departmentNode.innerText = department.name;
  departmentNode.value = department.name;
  jobDepartmentsFilter.appendChild(departmentNode);
}

function addListenertoDepartmentFilter() {
  let allFiltersNode = document.querySelector("[jobs-filters='departments']");
  allFiltersNode.addEventListener("change", async (e) => {
    selectedDepartment = e.target.value;
    const jobsFound = filterJobs(
      allJobs,
      selectedDepartment,
      selectedOfficeLocation
    );

    if (jobsFound.length > 0) {
      noJobsFoundNode.style.display = "none";
      await writeJobs(jobsFound);
      if (selectedOfficeLocation !== "All Offices") {
        const jobsNodes = jobsWrapper.querySelectorAll("a");
        jobsNodes.forEach((jobNode) => {
          if (jobNode.getAttribute("job-location") !== selectedOfficeLocation) {
            jobNode.style.display = "none";
          }
        });
      }
    } else {
      jobsWrapper.innerHTML = "";
      // no jobs found
      noJobsFoundNode.style.display = "block";
    }
  });
}

function renderOfficeLocations(office) {
  let officeNode = document.createElement("option");
  officeNode.innerText = office;
  officeLocationsFilter.appendChild(officeNode);
}

function addListenerToOfficeFilters() {
  let allFiltersNode = document.querySelector("[jobs-filters='locations']");
  allFiltersNode.addEventListener("change", async (e) => {
    selectedOfficeLocation = e.target.value;

    let jobsFound = filterJobs(
      allJobs,
      selectedDepartment,
      selectedOfficeLocation
    );

    if (jobsFound.length > 0) {
      noJobsFoundNode.style.display = "none";
      await writeJobs(jobsFound, selectedOfficeLocation, selectedDepartment);
      if (selectedOfficeLocation !== "All Offices") {
        const jobsNodes = jobsWrapper.querySelectorAll("a");
        jobsNodes.forEach((jobNode) => {
          if (jobNode.getAttribute("job-location") !== selectedOfficeLocation) {
            jobNode.style.display = "none";
          }
        });
      }
    } else {
      jobsWrapper.innerHTML = "";
      // no jobs found
      noJobsFoundNode.style.display = "block";
    }
  });
}

function filterJobs(array, selected_department, selected_location) {
  if (
    selected_department === "View All" &&
    selected_location === "All Offices"
  ) {
    // Case 1: Both selected_department and selected_location are "ALL"
    return array;
  } else if (
    selected_department !== "View All" &&
    selected_location === "All Offices"
  ) {
    // Case 2: Only selected_department has a value
    return array.filter(
      (dept) => dept.name.trim() === selected_department.trim()
    );
  } else if (
    selected_location !== "All Offices" &&
    selected_department === "View All"
  ) {
    // Case 3: Only selected_location has a value
    return array.filter((obj) =>
      obj.jobs.some((job) => job.location.name.includes(selected_location))
    );
  } else {
    // Case 4: Both selected_department and selected_location have values
    return array.filter(
      (obj) =>
        obj.name === selected_department &&
        obj.jobs.some((job) => job.location.name.includes(selected_location))
    );
  }
}
